std::vector<Vertex> vertices;

const uint32_t latitudeSegments = 32;
const uint32_t longitudeSegments = 64;

for (uint32_t i = 0; i <= latitudeSegments; ++i) {
    float theta = glm::pi<float>() * i / latitudeSegments;

    for (uint32_t j = 0; j <= longitudeSegments; ++j) {
        float phi = 2.0f * glm::pi<float>() * j / longitudeSegments;

        glm::vec3 normal = {
            sin(theta) * cos(phi),
            cos(theta),
            sin(theta) * sin(phi)
        };

        vertices.push_back({
            .position = normal,
            .normal = normal,
            .texCoord = {
                static_cast<float>(j) / longitudeSegments,
                static_cast<float>(i) / latitudeSegments
            }
        });
    }
}

------------


std::vector<uint32_t> indices;

for (uint32_t i = 0; i < latitudeSegments; ++i) {
    for (uint32_t j = 0; j < longitudeSegments; ++j) {
        uint32_t a = i * (longitudeSegments + 1) + j;
        uint32_t b = a + longitudeSegments + 1;

        indices.push_back(a);
        indices.push_back(b);
        indices.push_back(a + 1);

        indices.push_back(b);
        indices.push_back(b + 1);
        indices.push_back(a + 1);
    }
}

struct Mesh {
    std::vector<Vertex> vertices;
    std::vector<uint32_t> indices;
};
